\frohmantitle{A Lean-Verified Conditional Uniform Ceiling for a Cubic-Tethered Vorticity Model}

\begin{scopebox}
\textbf{Scope.} This paper concerns a \emph{modified} vorticity equation with an added cubic damping tether. That model is \textbf{not} Navier--Stokes. The main result is \emph{conditional} on the named Prop \lean{FieldStretchingLogLoss}, which is \textbf{not proved}, and on six further hypotheses. The paper does \textbf{not} solve the Clay Millennium Problem, and it says nothing about regularity of unmodified 3D Navier--Stokes.
\end{scopebox}

\begin{abstract}
In the Lean~4/Mathlib~\cite{Lean4,Mathlib2020} repository \file{metriplectic-world-model} (commit \file{2274a23}), we prove a uniform-in-time ceiling on $\sup\abs{\omega}$ for a vorticity model with cubic tether $-\kappa\abs{\omega}^2\omega$ (\lean{field_uniform_ceiling_logLoss_cubicTether}). It is conditional on a pointwise log-loss stretching Prop and on six regularity, support and sign hypotheses, which we list verbatim. We explain why the known Calder\'on--Zygmund estimate does not supply the stretching Prop. The axioms used are only \lean{propext}, \lean{Classical.choice} and \lean{Quot.sound}.
\end{abstract}

\section{Introduction}
Adding a cubic damping term to the vorticity equation produces a model in which the peak vorticity $Y(t)=\sup_x\abs{\omega(t,x)}$ satisfies a comparison inequality of the form $Y'\le CY^2-\kappa Y^3$, which yields a uniform-in-time ceiling. The difficulty lies in the vortex-stretching term, whose known bounds involve norms stronger than $Y$. In this paper we isolate exactly what is needed: we prove, in Lean~4, a uniform ceiling for the cubic-tethered model conditional on a $Y$-only logarithmic stretching hypothesis and six explicit regularity and support hypotheses, and we explain why the classical Calder\'on--Zygmund estimate does not supply the stretching hypothesis.

\section{The cubic-tethered vorticity model}
The model is \lean{ModelVorticityPDE}~$\nu\ \omega\ u\ T$:
\begin{equation}\label{eq:model}u=\BS(\omega),\qquad \Div\omega=0,\qquad \partial_t\omega+(u\cdot\nabla)\omega-(\omega\cdot\nabla)u=\nu\Delta\omega+T.\end{equation}
It uses the cubic tether $T=$\,\lean{cubicTether}~$\kappa$~$\omega$, i.e.\ $T=-\kappa\abs{\omega}^2\omega$. This is \emph{not} Navier--Stokes; the tether is an added damping term. Write
\[Y(t)=\sup_x\abs{\omega(t,x)},\qquad \mathrm{ceiling}(M_0,C,\kappa)=\max\bigl(M_0,\max(C/\kappa,0)\bigr).\]

\begin{hyp}[\lean{FieldStretchingLogLoss}~$\omega\ u\ Y\ C_1\ C_2$\coined; \textbf{not proved}]
For every $t\ge0$ and every global maximizer $x_*$ of $|\omega(t,\cdot)|$:
\begin{equation}\label{eq:logloss}\omega\cdot(\omega\cdot\nabla)u\,(t,x_*)\le C_1Y^3+C_2Y^3\log(1+Y).\end{equation}
\end{hyp}

\begin{theorem}[\lean{field_uniform_ceiling_logLoss_cubicTether}\coined; conditional]
Assume all of the following:
\begin{enumerate}
\item $\nu\ge0$, $\kappa>0$, $C_2\ge0$;
\item \lean{ModelVorticityPDE}~$\nu\ \omega\ u$ (\lean{cubicTether}~$\kappa$~$\omega$);
\item a \emph{fixed} compact nonempty set $K$ with $\omega(s,x)=0$ for $x\notin K$, for all $s$;
\item $\omega(t)\in C^2$ in space for every $t$;
\item joint continuity of $\omega$ and of $\partial_t\omega$, and the pointwise time derivative $\partial_t\omega(s,x)$ exists for all $s,x$;
\item $Y(s)$ is the least upper bound of $\{|\omega(s,x)|\}$ for all $s$;
\item \lean{FieldStretchingLogLoss}~$\omega\ u\ Y\ C_1\ C_2$.
\end{enumerate}
Then for all $t\ge0$,
\begin{equation}\label{eq:main}Y(t)\le\mathrm{ceiling}\Bigl(Y(0),\;C_1+\frac{\kappa}{4}+\frac{4C_2^2}{\kappa},\;\frac{3\kappa}{4}\Bigr).\end{equation}
\end{theorem}
\emph{Proof route.}
\begin{itemize}
\item \lean{fieldNetRate_of_logLoss_cubicTether}\coined{} uses $\log(1+y)\le2\sqrt{1+y}$ and Young's inequality to absorb the log loss into the quartic tether. This gives \lean{FieldNetRate} with $C=C_1+\kappa/4+4C_2^2/\kappa$ and $\kappa_{\rm eff}=3\kappa/4$.
\item \lean{field_uniform_ceiling_of_netRate}\coined{} applies a Danskin upper-Dini bound for $Y^2$ at a maximizer. There the transport term vanishes and $\omega\cdot\Delta\omega\le0$.
\item A Dini comparison lemma, \lean{ceiling_of_upper_dini}\coined, finishes the proof. No differentiability of $Y$ is assumed.
\end{itemize}

\paragraph{Why the known Calder\'on--Zygmund estimate does not suffice.}
The classical log estimate (Beale--Kato--Majda~\cite{BKM1984}, Kozono--Taniuchi~\cite{KozonoTaniuchi2000} type) reads
\begin{equation}\label{eq:bkmlog}\norm{\nabla u}_{L^\infty}\le C\Bigl(1+Y\bigl(1+\log^+(\norm{\omega}_{C^\alpha}/Y)\bigr)+\norm{\omega}_{L^2}\Bigr).\end{equation}
The logarithm contains a \emph{higher} norm, $\|\omega\|_{C^\alpha}$. The ceiling argument controls only $Y$, and a time-uniform bound on $\|\omega(t)\|_{C^\alpha}$ is essentially the regularity question itself. So the known estimate does not imply \lean{FieldStretchingLogLoss}, whose $Y$-only form is a strictly stronger hypothesis than anything known.

The CZ estimate itself is also not formalized in that repository, which has no singular-integral or H\"older theory for $\nabla\mathrm{BS}$. The existence of solutions with uniformly fixed compact support (hypothesis 3) is likewise not established. Neither the theorem nor anything else in this paper says anything about unmodified Navier--Stokes.
