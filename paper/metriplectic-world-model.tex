\begin{center}\fbox{\parbox{0.9\textwidth}{\textbf{Draft --- tracks build in progress.} This repository formalizes a \emph{model}: the tethered metriplectic system, which adds a symmetric tether bracket and cubic damping of peak vorticity to 3D vorticity dynamics. It proves nothing about unmodified Navier--Stokes ($\kappa=0$) and does not resolve the Clay Millennium problem.}}\end{center}
\begin{abstract}
We report a Lean~4/Mathlib formalization (0 \texttt{sorry}; axioms only \lean{propext}, \lean{Classical.choice}, \lean{Quot.sound}) of the author's tethered metriplectic model. Proved unconditionally: the unique antisymmetric/symmetric splitting of a bilinear bracket (\lean{split_exists_unique}); a non-circular canonicity theorem forcing the metric bracket to the form $c\langle\Pi_u x,\Pi_u y\rangle$ (\lean{canonical_metric_bracket}); a uniform vorticity ceiling for the comparison inequality $M'\le CM^2-\kappa M^3$; a log-loss replacement for the stretching hypothesis giving a ceiling when $\kappa>4C_{\mathrm{abs}}$; several encoding counterexamples; the Newton-potential identity $\Delta(N\star g)=-4\pi g$ on $\mathbb{R}^3$ for $g\in C^2_c$; and $\operatorname{curl}\mathrm{BiotSavart}\,\omega=\omega$ for smooth, compactly supported, divergence-free $\omega$ (\lean{curlBiotSavartTest}). Conditional and open items are listed by their exact Prop names.
\end{abstract}
\section{Naming}
\lean{Pi_u} is the author's own name (manuscript of 31 May 2026): the projection $\Pi_u x = x-(\langle u,x\rangle/\|u\|^2)u$ off $\operatorname{span}\{u\}$. All other identifiers defined in this repository are \emph{coined in this build} (marked \coined), including \lean{canonical_metric_bracket}, kept by the author's choice and flagged as coined. Modules under \texttt{FrohmanianSymplecticTether/} and \texttt{TetherMayForm/} are ported verbatim from earlier repositories.
\section{Manuscript framing}
The May/June 2026 manuscript introduces a tethered Lyapunov functional $S_\varepsilon$ and reduces regularity to the comparison ODE $y'=Cy^2-\kappa''y^3$ for the peak vorticity, through transport cancellation, viscous dissipation at the maximum, a H\"older--Sobolev gate, and Young absorption at $\varepsilon=\kappa/4$ (Proposition~2). Within the model, the cubic term comes from the tether. The manuscript's hypothesis (H5), a non-circular majorant for the vorticity, is the genuine gap; for $\kappa=0$ it is the Clay core. Here $\kappa$ has units of time and breaks Navier--Stokes scaling, which is why the model is distinct from NS.
\section{Brackets and canonicity (\texttt{Bracket.lean}, \texttt{Canonical.lean})}
\begin{theorem}\lean{split_exists_unique}\coined: every bilinear bracket is uniquely an antisymmetric part plus a symmetric part.\end{theorem}
\begin{theorem}\lean{canonical_metric_bracket}\coined: on a real inner product space, a bilinear form $B$ that is symmetric (C1), degenerate along $u$ (C2: $B(u,\cdot)=0$) and invariant under every linear isometry fixing $u$ equals $c\,\langle \Pi_u x,\Pi_u y\rangle$. The hypotheses do not mention $\Pi_u$. \lean{canonical_metric_bracket_sign}\coined: if $B(x,x)\le0$ then $c\le0$. \lean{Pi_u_form_symm_degen}\coined: the forced form satisfies C1, C2.\end{theorem}
\noindent\emph{Scope:} pointwise at one state; does not force the $|\omega|^2$ weight, the value of $\kappa$, or C3. Also proved: \lean{energy_conserved}, \lean{energy_const_along}, \lean{entropy_production}, \lean{gen_entropy_eq}, \lean{entropy_monotone_along}\coined.
\section{Ceiling and stretching (\texttt{Ceiling.lean}, \texttt{AuditFixes.lean})}
\begin{theorem}\lean{uniform_ceiling}\coined: if $M'\le CM^2-\kappa M^3$, $\kappa>0$, then $M(t)\le\max(M(0),\max(C/\kappa,0))$ for $t\ge0$. Also \lean{absorbing}, \lean{ceiling_antitone_kappa}, \lean{kappa_zero_no_ceiling}\coined{} (at $\kappa=0$, $1/(1-t)$ satisfies $M'\le M^2$ and blows up).\end{theorem}
\paragraph{Correction.} The hypothesis \lean{StretchingCZ} (a pure $y^3$ bound on stretching at the maximum) is \emph{false as a sup-norm bound} for Navier--Stokes: the known bound carries a logarithmic loss. Results using it are relabeled as model-only. Its honest replacement is \lean{StretchingLogLoss}~$d\,C_1\,C_2$\coined: $St\le C_1y^3+C_2y^3\log(1+y)$.
\begin{theorem}\lean{maxpoint_rate_logLoss}\coined: under \lean{EnergyIdentity}, \lean{StretchingLogLoss}, \lean{ViscousAtMax}, \lean{TetherFeedback}, \lean{CrossControlled},
\[y'\le\Bigl(C_1+\tfrac{\kappa}{4}+\tfrac{4C_2^2}{\kappa}\Bigr)y^2-\Bigl(\tfrac{\kappa}{4}-C_{\mathrm{abs}}\Bigr)y^3,\]
so with \lean{uniform_ceiling} a uniform ceiling follows whenever $\kappa>4C_{\mathrm{abs}}$ (Young absorption at $\varepsilon=\kappa/4$).\end{theorem}

\section{Newton potential on $\mathbb{R}^3$ (\texttt{NewtonKernel}, \texttt{RegularizedKernel}, \texttt{NewtonLaplacian})}
Proved: \lean{scalarLaplacian_newtonRadial} ($\Delta(1/|y|)=0$, $y\ne0$), \lean{contDiff_newtonPotential}, \lean{hasFDerivAt_newtonPotential} (differentiation under the integral); for $N_\varepsilon=(|y|^2+\varepsilon^2)^{-1/2}$: \lean{scalarLaplacian_regNewton} ($\Delta N_\varepsilon=-3\varepsilon^2(|y|^2+\varepsilon^2)^{-5/2}$), \lean{regNewton_le}, \lean{tendsto_regNewton}, \lean{integral_neg_scalarLaplacian_regNewton} ($\int-\Delta N_\varepsilon=4\pi$), \lean{tendsto_regPeak_approxIdentity} (all \coined). The regularized route avoids spherical Green identities, which Mathlib lacks.
Step 3 is closed (\texttt{NewtonLaplacian}): \lean{integral_mul_scalarLaplacian_eq} ($\int K\,\Delta g=\int \Delta K\,g$ for $K\in C^2$, $g\in C^2_c$, by Mathlib integration by parts, no boundary terms), \lean{regNewton_conv_laplacian} ($N_\varepsilon\star\Delta g=(\Delta N_\varepsilon)\star g$), \lean{tendsto_regNewton_conv} (the $\varepsilon\to0$ limit, dominated by $N|h|$), \lean{scalarLaplacian_newtonPotential} ($\Delta(N\star g)=N\star\Delta g$), and
\begin{theorem}\lean{scalarLaplacian_newtonPotential_eq}\coined: for $g\in C^2_c(\mathbb{R}^3)$ and $N(y)=1/|y|$, $\Delta(N\star g)(x)=-4\pi g(x)$ for every $x$.\end{theorem}
\section{Biot--Savart on the test class (\texttt{BiotSavartNewton})}
With $G_j(t)=t_j/|t|^3$ (\lean{gradKer}\coined{} at $\varepsilon=0$):
\begin{theorem}\lean{newtonPotential_deriv_eq}\coined: for $h\in C^1_c$, $N\star\partial_jh=-\int G_j(t)\,h(x-t)\,dt$. (Regularized identity \lean{regNewton_conv_deriv}, then $\varepsilon\to0$ on both sides; the right side is dominated by $|t|^{-2}|h|$.)\end{theorem}
\begin{theorem}[Step 5]\lean{biotSavart_eq_curl_newtonVec}\coined: for \lean{IsTestOrbit}~$\omega$, $\mathrm{BiotSavart}\,\omega=(4\pi)^{-1}\operatorname{curl}(N\star\omega)$ pointwise, $N\star\omega$ taken componentwise (\lean{newtonVec}).\end{theorem}
\begin{theorem}[Step 4]\lean{curl_curl_newtonVec}\coined: for \lean{IsTestOrbit}~$\omega$, $\operatorname{curl}\operatorname{curl}(N\star\omega)=4\pi\omega$, using symmetry of second derivatives (\lean{fderiv_fderiv_swap}), $\operatorname{div}(N\star\omega)=N\star\operatorname{div}\omega=0$, and the step-3 theorem.\end{theorem}
\begin{theorem}\lean{curlBiotSavartTest}\coined: \lean{CurlBiotSavartTest} holds, i.e.\ $\operatorname{curl}\mathrm{BiotSavart}\,\omega=\omega$ for every smooth, compactly supported, divergence-free $\omega$. Consequently \lean{helicity_casimir_test_closed}\coined: helicity is a Casimir of the Arnold bracket at such $\omega$, given the explicit argument $D\,S=u$.\end{theorem}
\noindent\emph{Scope:} this is the classical identity on the smooth compactly supported class only; it says nothing about the dynamics. That $2u$ is the true G\^ateaux derivative of helicity (Biot--Savart self-adjointness) is not proved here.
\section{Status}
\begin{itemize}
\item \textbf{Proved unconditionally:} \lean{split_exists_unique}, \lean{canonical_metric_bracket} (+ sign, symm/degen), energy/entropy structure, \lean{uniform_ceiling}, \lean{maxpoint_rate_logLoss} (as an implication from its listed Props), the obstructions above, the Newton potential identity (steps 1--3), \lean{curlBiotSavartTest} (steps 4--5), the scalar instance \lean{instance_global_regularity}.
\item \textbf{Conditional on named Props:} \lean{model_uniform_ceiling} (\lean{TetheredMaxRate}); \lean{model_global_regularity} (\lean{TetheredMaxRate}, \lean{ModelContinuation}); \lean{model_global_regularity_of_realization} (\lean{MaxPointRealization}, whose fields include \lean{StretchingCZ} --- false as an $L^\infty$ NS bound); velocity-tether energy/entropy (\lean{KineticEnergyCurlIdentity}, \lean{EntropyCasimir}); helicity Casimir on the test class given $D\,S=u$ (\lean{helicity_casimir_test_closed}; no longer conditional on \lean{CurlBiotSavartTest}).
\item \textbf{Open:} that $D\,S=2u$ is the G\^ateaux derivative of helicity; a.e.\ uniqueness modulo gradients (\lean{StreamFunctionRepModGradOn}); \lean{EntropyCasimir} for the concrete bracket; deriving the $|\omega|^2$ weight, $\kappa$ and C3 from C1--C3; the link of maximizer data to an actual vorticity field; and, for unmodified NS, the (H5) majorant, i.e.\ the Clay core. Neither the Clay problem nor NS regularity is claimed.
\item The type named \lean{T3} is $\mathbb{R}^3$, not the torus.
\end{itemize}
